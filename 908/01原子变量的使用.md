```C++
#include <atomic>
#include <thread>
#include <iostream>
#include <vector>

std::atomic<int> counter(0);  // 原子整型，天生线程安全

void atomic_increment() {
    for (int i = 0; i < 100000; i++) {
        counter++;  // ✅ 这是原子操作，无需加锁
    }
}

int main() {
    std::vector<std::thread> threads;
    for (int i = 0; i < 10; i++) {
        threads.emplace_back(atomic_increment);
    }
    for (auto& t : threads) t.join();
  
    std::cout << "最终结果: " << counter << std::endl;  // 完美输出 1000000
    return 0;
}
```
