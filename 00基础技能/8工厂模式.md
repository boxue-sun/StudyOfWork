// ============================================================
// 工厂模式（Factory Pattern）面试手写示例
// 包含：简单工厂（Simple Factory）、工厂方法（Factory Method）
// 考点：多态、虚析构、unique_ptr、开闭原则
// ============================================================

#include <iostream></iostream>
#include <memory></memory>   // unique_ptr, make_unique

// ========== 1. 抽象产品（Product） ==========
// 所有产品的基类，定义统一接口。
// 面试点：基类必须要有虚析构函数！否则用基类指针 delete 派生类对象时，
// 只会调用基类析构，派生类资源泄漏，属于未定义行为（UB）。
class Product {
public:
    virtual ~Product() = default;          // 虚析构，保证多态删除安全
    virtual void use() const = 0;          // 纯虚函数，子类必须实现
};

// ========== 2. 具体产品 A（ConcreteProductA） ==========
class ConcreteProductA : public Product {
public:
    void use() const override {
        std::cout << "使用产品 A" << std::endl;
    }
};

// ========== 3. 具体产品 B（ConcreteProductB） ==========
class ConcreteProductB : public Product {
public:
    void use() const override {
        std::cout << "使用产品 B" << std::endl;
    }
};

// ========== 4. 简单工厂（Simple Factory） ==========
// 一个工厂类根据参数创建不同产品。
// 优点：客户端不需要知道具体产品类名，只需传枚举/字符串。
// 缺点：新增产品必须修改工厂类，违反“开闭原则”。
// 适用：产品种类少、不常变动的场景。
class SimpleFactory {
public:
    // 用枚举代替字符串：避免拼写错误，效率更高，编译器可检查。
    enum Type { A, B };

    // 静态方法，直接通过类名调用：SimpleFactory::create(...)
    // 返回 unique_ptr：自动管理内存，避免手动 delete，异常安全。
    static std::unique_ptr<Product></product> create(Type t) {
        switch (t) {
            case A:
                // make_unique 是 C++14 特性；C++11 可写：
                // return std::unique_ptr<Product></product>(new ConcreteProductA());
                return std::make_unique<ConcreteProductA></concreteproducta>();
            case B:
                return std::make_unique<ConcreteProductB></concreteproductb>();
            default:
                return nullptr;            // 防御性编程：未知类型返回空
        }
    }
};

// ========== 5. 工厂方法（Factory Method） ==========
// 抽象工厂：定义创建产品的接口，但由子类决定实例化哪个产品。
// 优点：符合开闭原则——新增产品只需新增工厂子类，不修改原有代码。
// 缺点：类数量增加（每个产品对应一个工厂）。
class Factory {
public:
    virtual ~Factory() = default;          // 同样需要虚析构
    virtual std::unique_ptr<Product></product> createProduct() const = 0;
};

// 具体工厂 A：专门生产产品 A
class FactoryA : public Factory {
public:
    std::unique_ptr<Product></product> createProduct() const override {
        return std::make_unique<ConcreteProductA></concreteproducta>();
    }
};

// 具体工厂 B：专门生产产品 B
class FactoryB : public Factory {
public:
    std::unique_ptr<Product></product> createProduct() const override {
        return std::make_unique<ConcreteProductB></concreteproductb>();
    }
};

// ========== 6. 测试代码 ==========
int main() {
    // ---------- 简单工厂使用 ----------
    // 客户端只需知道枚举值，完全解耦具体产品类。
    auto p1 = SimpleFactory::create(SimpleFactory::A);
    auto p2 = SimpleFactory::create(SimpleFactory::B);

    if (p1) p1->use();   // 输出：使用产品 A
    if (p2) p2->use();   // 输出：使用产品 B

    // ---------- 工厂方法使用 ----------
    // 面向抽象工厂编程，可以方便地替换具体工厂。
    std::unique_ptr<Factory></factory> factory = std::make_unique<FactoryB></factoryb>();
    auto p3 = factory->createProduct();
    if (p3) p3->use();   // 输出：使用产品 B

    // 想换产品？只需换工厂实例，无需修改客户端逻辑：
    // factory = std::make_unique<FactoryA></factorya>();
    // auto p4 = factory->createProduct();
    // p4->use();        // 输出：使用产品 A

    return 0;
}

	一个类，里面的一个成员函数实现创建具体产品的实例对象
