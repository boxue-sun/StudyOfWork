+bool IsLineIntersect(const std::list<Point2d>& line1, const std::list<Point2d>& line2) {
+    if (line1.size() < 2 || line2.size() < 2) {
+        return false;
+    }
+    for (auto line1_point1 = line1.begin(); std::next(line1_point1) != line1.end(); ++line1_point1) {
+        for (auto line2_point1 = line2.begin(); std::next(line2_point1) != line2.end(); ++line2_point1) {
+            /* Whether line segment is intersected. */
+            auto line1_point2 = std::next(line1_point1);
+            auto line2_point2 = std::next(line2_point1);
+            float cross_product_1 = (line1_point2->x - line1_point1->x) * (line2_point2->y - line2_point1->y) -
+                                    (line2_point2->x - line2_point1->x) * (line1_point2->y - line1_point1->y);
+            if (fabs(cross_product_1 - 0) < std::numeric_limits<float>::epsilon()) {
+                continue;  // parallel
+            }
+            float cross_product_2 = (line2_point2->x - line2_point1->x) * (line1_point1->y - line2_point1->y) -
+                                    (line1_point1->x - line2_point1->x) * (line2_point2->y - line2_point1->y);
+            float ratio_1 = cross_product_2 / cross_product_1;
+            float cross_product_3 = (line1_point2->x - line1_point1->x) * (line1_point1->y - line2_point1->y) -
+                                    (line1_point1->x - line2_point1->x) * (line1_point2->y - line1_point1->y);
+            float ratio_2 = cross_product_3 / cross_product_1;
+            if (ratio_1 < 0 || ratio_1 > 1 || ratio_2 < 0 || ratio_2 > 1) {
+                continue;  // not intersected
+            }
+            return true;  // intersected
+        }
+    }
+    return false;
+}
+
+bool IsPointOnBoundary(const Point2d& point, const std::vector<Point2d>& boundary) {
+    if(boundary.size() < 3){
+        std::cerr << "Warning, boundary points less than 3!" << std::endl;
+        return false;
+    }
+    std::vector<Segment2d> segments;
+    segments.reserve(boundary.size());
+    for (size_t i = 0; i < boundary.size(); ++i) {
+        segments.emplace_back(boundary[i], boundary[(i + 1) % boundary.size()]);
+    }
+    return std::any_of(segments.begin(), segments.end(), [&](const Segment2d& poly_seg) { return IsPointIn(point, poly_seg); });
+}
+
+bool IsPointInPolygon(const Point2d& point, const std::vector<Point2d>& polygon) {
+    if (polygon.size() < 3) {
+        std::cerr << "Warning, polygon points less than 3!" << std::endl;
+        return false;
+    }
+    if(IsPointOnBoundary(point, polygon)){
+        return true;
+    }
+    int j = polygon.size() - 1;
+    int c = 0;
+    for (int i = 0; i < polygon.size(); ++i) {
+        if((polygon[i].y > point.y) != (polygon[j].y > point.y)){
+            const double side = (polygon[i].x - polygon[j].x) * (point.y - polygon[j].y) - (polygon[i].y - polygon[j].y) * (point.x - polygon[j].x);
+            if(polygon[i].y < polygon[j].y ? side > 0.0: side < 0.0){
+                ++c;
+            }
+        }
+        j = i;
+    }
+    return c & 1;
+}
